   119→function b64d(s) {
   120→  try {
   121→    let t = s.replace(/-/g, "+").replace(/_/g, "/").trim();
   122→    while (t.length % 4) t += "=";
   123→    const buf = Buffer.from(t, "base64");
   124→    if (buf.length === 0) return "";
   125→    const str = buf.toString("utf8");
   126→    return /^[\x20-\x7e\s]/.test(str.slice(0, 8)) ? str : "";
   127→  } catch {
   128→    return "";
   129→  }
   130→}
   131→function safeDecode(s) {
   132→  try {
   133→    return decodeURIComponent(s);
   134→  } catch {
   135→    return s;
   136→  }

   140→function parseVmess(uri) {
   141→  const json = b64d(uri.slice(8));
   142→  if (!json) return null;
   143→  try {
   144→    const j = JSON.parse(json);
   145→    if (!j.add || !j.port) return null;
   146→    return {
   147→      raw: uri,
   148→      proto: "vmess",
   149→      host: String(j.add),
   150→      port: Number(j.port),
   151→      name: String(j.ps || ""),
   152→      net: String(j.net || "tcp"),
   153→      security: String(j.tls) === "tls" ? "tls" : "none",
   154→    };
   155→  } catch {
   156→    return null;

   160→function parseUserinfo(uri, proto) {
   161→  const m = uri.match(
   162→    new RegExp("^" + proto + "://[^@]+@([^:/#]+):(\\d+)(?:/[^?]*)?(?:\\?([^#]*))?(?:#(.*))?$")
   163→  );
   164→  if (!m) return null;
   165→  const q = {};
   166→  if (m[3])
   167→    for (const kv of m[3].split("&")) {
   168→      const i = kv.indexOf("=");
   169→      if (i > 0) q[kv.slice(0, i)] = safeDecode(kv.slice(i + 1));
   170→    }
   171→  return {
   172→    raw: uri,
   173→    proto,
   174→    host: m[1],
   175→    port: Number(m[2]),
   176→    name: safeDecode(m[4] || ""),
   177→    net: q.type || q.network || "tcp",
   178→    security: q.security || (q.tls === "1" ? "tls" : "none"),
   179→  };
   180→}

   182→function parseSs(uri) {
   183→  let m = uri.match(/^ss:\/\/([^@#]+)@([^:]+):(\d+)(?:\?[^#]*)?#(.*)$/);
   184→  if (m)
   185→    return {
   186→      raw: uri,
   187→      proto: "ss",
   188→      host: m[2],
   189→      port: Number(m[3]),
   190→      name: safeDecode(m[4] || ""),
   191→      net: "tcp",
   192→      security: "none",
   193→    };
   194→  m = uri.match(/^ss:\/\/([^#]+?)(?:#(.*))?$/);
   195→  if (m) {
   196→    const dec = b64d(m[1]);
   197→    const mm = dec.match(/@([^:]+):(\d+)$/);
   198→    if (!mm) return null;
   199→    return {
   200→      raw: uri,
   201→      proto: "ss",
   202→      host: mm[1],
   203→      port: Number(mm[2]),
   204→      name: safeDecode(m[2] || ""),
   205→      net: "tcp",
   206→      security: "none",
   207→    };
   208→  }
   209→  return null;
   210→}

   238→const URI_RE = /^(vmess|vless|trojan|ss):\/\//i;
   239→function extractUris(text, uris) {
   240→  for (const line of text.split(/\r?\n/)) {
   241→    const t = line.trim();
   242→    if (!t) continue;
   243→    if (URI_RE.test(t)) {
   244→      uris.add(t);
   245→      continue;
   246→    }
   247→    const dec = b64d(t);
   248→    if (dec && URI_RE.test(dec.trim())) uris.add(dec.trim());
   249→  }
   250→}

   265→function parseAll(uris) {
   266→  const cands = [];
   267→  const seen = new Set();
   268→  for (const uri of uris) {
   269→    const low = uri.slice(0, 8).toLowerCase();

   273→    else if (low.startsWith("trojan://")) c = parseUserinfo(uri, "trojan");
   274→    else if (low.startsWith("ss://")) c = parseSs(uri);
   275→    if (!c || !c.host || !c.port || c.port > 65535) continue;
   276→    const key = `${c.proto}|${c.host.toLowerCase()}|${c.port}`;
   277→    if (seen.has(key)) continue;
   278→    seen.add(key);
   279→    cands.push(c);
   280→  }
   281→  return cands;
   282→}

   285→async function resolveHosts(cands) {
   286→  const resolver = new Resolver();
   287→  const deFirst = cands.filter((c) => remarkToCc(c.name));
   288→  const others = cands.filter((c) => !remarkToCc(c.name));
   289→  const hosts = [];
   290→  const seen = new Set();
   291→  for (const c of [...deFirst, ...others]) {
   292→    const k = c.host.toLowerCase();
   293→    if (seen.has(k)) continue;
   294→    seen.add(k);
   295→    hosts.push(c.host);
   296→    if (hosts.length >= MAX_HOSTS_RESOLVE) break;
   297→  }
   298→  const ipToHosts = new Map();
   299→  let idx = 0;
   300→  const CONC = 60;
   301→  const worker = async () => {
   302→    while (idx < hosts.length) {
   303→      const host = hosts[idx++];
   304→      try {
   305→        const addrs = await resolver.resolve4(host, { ttl: 300 });
   306→        for (const a of addrs.slice(0, 3)) {
   307→          if (!ipToHosts.has(a)) ipToHosts.set(a, []);
   308→          ipToHosts.get(a).push(host.toLowerCase());
   309→        }
   310→      } catch {
   311→        /* رزول نشد */
   312→      }
   313→    }
   314→  };
   315→  await Promise.all(Array.from({ length: CONC }, worker));
   316→  return ipToHosts;
   317→}

   319→async function geoBatch(ipToHosts) {
   320→  const ipList = [...ipToHosts.keys()].slice(0, MAX_GEO_IPS);
   321→  const geo = new Map();
   322→  let healthy = true;
   323→  // از curl استفاده می‌کنیم چون fetch در برخی محیط‌ها (مانند bun در sandbox) معلق می‌شود
   324→  const curlBatch = (body) => {
   325→    try {
   326→      const out = execFileSync(
   327→        "curl",
   328→        [
   329→          "-sS", "--max-time", "20", "-X", "POST",
   330→          "-H", "Content-Type: application/json",
   331→          "-d", body,
   332→          "http://ip-api.com/batch?fields=countryCode,org,query",
   333→        ],
   334→        { timeout: 25000, maxBuffer: 10 * 1024 * 1024 }
   335→      );
   336→      return JSON.parse(out.toString());
   337→    } catch {
   338→      return null;
   339→    }
   340→  };
   341→  for (let i = 0; i < ipList.length && healthy; i += 100) {
   342→    const chunk = ipList.slice(i, i + 100);
   343→    let ok = false;
   344→    for (let attempt = 0; attempt < 2 && !ok; attempt++) {
   345→      const arr = curlBatch(JSON.stringify(chunk));
   346→      if (Array.isArray(arr)) {
   347→        for (const r of arr)
   348→          if (r && r.query)
   349→            geo.set(r.query, { cc: r.countryCode || "", org: r.org || r.isp || "" });
   350→        ok = true;
   351→      } else {
   352→        await new Promise((r) => setTimeout(r, 5000));
   353→      }
   354→    }
   355→    if (!ok) {
   356→      if (geo.size === 0) healthy = false; // geo کلا در دسترس نیست
   357→      break;
   358→    }
   359→    await new Promise((r) => setTimeout(r, 4300));
   360→  }
   361→  return { geo, healthy };
   362→}

   365→async function tcpAlive(cands) {
   366→  const alive = [];
   367→  let idx = 0;
   368→  const CONC = 50;
   369→  const tcpCheck = (host, port) =>
   370→    new Promise((resolve) => {
   371→      let settled = false;
   372→      let s;
   373→      const done = (ok) => {
   374→        if (settled) return;
   375→        settled = true;
   376→        try {
   377→          s.destroy();
   378→        } catch {}
   379→        resolve(ok);
   380→      };
   381→      try {
   382→        s = net.connect({ host, port });
   383→      } catch {
   384→        return resolve(false);
   385→      }
   386→      s.setTimeout(3000, () => done(false));
   387→      s.once("connect", () => done(true));
   388→      s.once("error", () => done(false));
   389→    });
   390→  const worker = async () => {
   391→    while (idx < cands.length) {
   392→      const c = cands[idx++];
   393→      if (await tcpCheck(c.host, c.port)) alive.push(c);
   394→    }
   395→  };
   396→  await Promise.all(Array.from({ length: CONC }, worker));
   397→  return alive;
   398→}

   409→  if (network === "ws") s.wsSettings = { path: p, headers: h ? { Host: h } : undefined };
   410→  else if (network === "grpc") s.grpcSettings = { serviceName: p.replace(/^\//, "") || "grpc" };
   411→  else if (network === "httpupgrade") s.httpupgradeSettings = { path: p, host: h || host };
   412→  else if (network === "xhttp") s.xhttpSettings = { path: p, host: h || host, mode: "auto" };
   413→  else if (network === "http") s.httpSettings = { path: p, host: h ? [h] : undefined };
   414→  else if (network === "tcp" && String(j.type) === "http")
   415→    s.tcpSettings = {
   416→      header: {
   417→        type: "request",
   418→        request: {
   419→          version: "1.1",
   420→          method: "GET",
   421→          path: [p],
   422→          headers: { Host: [h || host], "User-Agent": ["Mozilla/5.0"] },
   423→        },
   424→      },
   425→    };
   426→  return s;

   429→function streamFromQuery(q, host, defaultTls) {
   430→  const netMap = { raw: "tcp", splithttp: "xhttp" };
   431→  const network = netMap[String(q.type || "tcp").toLowerCase()] || String(q.type || "tcp").toLowerCase();
   432→  const sec = q.security || (defaultTls ? "tls" : "none");
   433→  const s = { network, security: sec };
   434→  if (sec === "tls")
   435→    s.tlsSettings = { serverName: q.sni || q.host || q.peer || host, fingerprint: q.fp || "chrome" };
   436→  else if (sec === "reality")
   437→    s.realitySettings = {
   438→      serverName: q.sni || host,
   439→      publicKey: q.pbk || "",
   440→      shortId: q.sid || "",
   441→      fingerprint: q.fp || "chrome",
   442→    };
   443→  const p = safeDecode(q.path || "/");
   444→  const h = q.host || "";
   445→  if (network === "ws") s.wsSettings = { path: p, headers: h ? { Host: h } : undefined };
   446→  else if (network === "grpc") s.grpcSettings = { serviceName: (q.serviceName || "").replace(/^\//, "") || "grpc" };
   447→  else if (network === "httpupgrade") s.httpupgradeSettings = { path: p, host: h || host };
   448→  else if (network === "xhttp") s.xhttpSettings = { path: p, host: h || host, mode: "auto" };
   449→  else if (network === "http") s.httpSettings = { path: p, host: h ? [h] : undefined };
   450→  else if (network === "tcp" && q.headerType === "http")
   451→    s.tcpSettings = {
   452→      header: { type: "request", request: { version: "1.1", method: "GET", path: [p], headers: { Host: [h || host] } } },
   453→    };
   454→  return s;
   455→}

   459→  if (low.startsWith("vmess://")) {
   460→    const j = JSON.parse(b64d(c.raw.slice(8)));
   461→    return {
   462→      protocol: "vmess",
   463→      settings: {
   464→        vnext: [{ address: c.host, port: c.port, users: [{ id: String(j.id), security: String(j.scy || "auto"), level: 0 }] }],
   465→      },
   466→      streamSettings: streamFromVmess(j, c.host),
   467→    };
   468→  }
   469→  if (low.startsWith("vless://")) {
   470→    const m = c.raw.match(/^vless:\/\/([^@]+)@([^:/#]+):(\d+)(?:\?([^#]*))?(?:#(.*))?$/);
   471→    if (!m) return null;
   472→    const q = {};
   473→    if (m[4])
   474→      for (const kv of m[4].split("&")) {
   475→        const i = kv.indexOf("=");
   476→        if (i > 0) q[kv.slice(0, i)] = safeDecode(kv.slice(i + 1));
   477→      }
   478→    return {
   479→      protocol: "vless",
   480→      settings: { vnext: [{ address: c.host, port: c.port, users: [{ id: m[1], encryption: "none", flow: q.flow || "", level: 0 }] }] },
   481→      streamSettings: streamFromQuery(q, c.host, false),
   482→    };
   483→  }
   484→  if (low.startsWith("trojan://")) {
   485→    const m = c.raw.match(/^trojan:\/\/([^@]+)@([^:/#]+):(\d+)(?:\?([^#]*))?(?:#(.*))?$/);
   486→    if (!m) return null;
   487→    const q = {};
   488→    if (m[4])
   489→      for (const kv of m[4].split("&")) {
   490→        const i = kv.indexOf("=");
   491→        if (i > 0) q[kv.slice(0, i)] = safeDecode(kv.slice(i + 1));
   492→      }
   493→    return {
   494→      protocol: "trojan",
   495→      settings: { servers: [{ address: c.host, port: c.port, password: safeDecode(m[1]) }] },
   496→      streamSettings: streamFromQuery(q, c.host, q.security !== "none"),
   497→    };
   498→  }
   499→  if (low.startsWith("ss://")) {
   500→    let userinfo = "";
   501→    const m1 = c.raw.match(/^ss:\/\/([^@#]+)@/);
   502→    if (m1) userinfo = b64d(m1[1]);
   503→    else {
   504→      const m2 = c.raw.match(/^ss:\/\/([^#]+)/);
   505→      const dec = b64d(m2 ? m2[1] : "");
   506→      const mi = dec.indexOf("@");
   507→      userinfo = mi > 0 ? dec.slice(0, mi) : "";
   508→    }
   509→    const ci = userinfo.indexOf(":");
   510→    if (ci < 0) return null;
   511→    return {
   512→      protocol: "shadowsocks",
   513→      settings: {
   514→        servers: [{ address: c.host, port: c.port, method: userinfo.slice(0, ci), password: userinfo.slice(ci + 1) }],
   515→      },
   516→    };
   517→  }
   518→  return null;
   519→}

   526→async function verifyOne(c, slot, expectedCc) {
   527→  let ob = null;
   528→  try {
   529→    ob = buildOutbound(c);
   530→  } catch {
   531→    return false;
   532→  }
   533→  if (!ob) return false;
   534→  const cfg = {
   535→    log: { loglevel: "warning" },
   536→    inbounds: [{ listen: "127.0.0.1", port: XRAY_BASE_PORT + slot, protocol: "socks", settings: { auth: "noauth", udp: false } }],
   537→    outbounds: [ob],
   538→  };
   539→  const cfgFile = path.join(TMP_DIR, `cfg_${slot}.json`);
   540→  fs.writeFileSync(cfgFile, JSON.stringify(cfg));
   541→  const proc = startXray(cfgFile);
   542→  const port = XRAY_BASE_PORT + slot;
   543→  await new Promise((r) => setTimeout(r, 1100));
   544→  const cc = await curlThroughSocks(port);
   545→  try {
   546→    proc.kill("SIGKILL");
   547→  } catch {}
   548→  if (!cc) return false;
   549→  if (expectedCc) return cc === expectedCc;
   550→  return !["IR", "CN", "KP", "SY"].includes(cc);
   551→}

   576→async function verifyBatch(list, expectedCc) {
   577→  const verified = new Set();
   578→  if (list.length === 0) return verified;
   579→  let idx = 0;
   580→  const N = Math.min(PARALLEL_XRAY, list.length);
   581→  const freeSlots = [...Array(N).keys()];
   582→  const takeSlot = () =>
   583→    new Promise((res) => {
   584→      const t = setInterval(() => {
   585→        if (freeSlots.length > 0) {
   586→          clearInterval(t);
   587→          res(freeSlots.shift());
   588→        }
   589→      }, 30);
   590→    });
   591→  const worker = async () => {
   592→    while (idx < list.length) {
   593→      const c = list[idx++];
   594→      const slot = await takeSlot();
   595→      try {
   596→        const ok = await verifyOne(c, slot, expectedCc);
   597→        if (ok) verified.add(c.raw);
   598→      } catch {
   599→        /* نادیده */
   600→      }
   601→      freeSlots.push(slot);
   602→    }
   603→  };
   604→  await Promise.all(Array.from({ length: N }, worker));
   605→  return verified;
   606→}

   609→function score(c) {
   610→  let s = 0;
   611→  if (PREF_PORTS.has(c.port)) s += 4;
   612→  if (c.security === "reality") s += 3;
   613→  if (c.security === "tls") s += 2;
   614→  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(c.host)) s += 1; // IP مستقیم
   615→  return s;
   616→}
   617→function pickForVerify(pool, n) {
   618→  const byProto = new Map();
   619→  const chosen = [];
   620→  const sorted = [...pool].sort((a, b) => score(b) - score(a));
   621→  for (const c of sorted) {
   622→    const k = `${c.proto}/${c.security}/${c.net}`;
   623→    const cnt = byProto.get(k) || 0;
   624→    if (cnt >= Math.ceil(n / 4) && chosen.length > 0) continue;
   625→    byProto.set(k, cnt + 1);
   626→    chosen.push(c);
   627→    if (chosen.length >= n) break;
   628→  }
   629→  return chosen;
   630→}

   634→async function ensureXray() {
   635→  if (fs.existsSync(XRAY_BIN) && fs.existsSync(path.join(path.dirname(XRAY_BIN), "geoip.dat"))) return true;
   636→  if (process.env.SKIP_VERIFY) return false;

   639→    const res = await fetch("https://github.com/XTLS/Xray-core/releases/latest/download/Xray-linux-64.zip", { signal: AbortSignal.timeout(180000) });
   640→    if (!res.ok) throw new Error(`HTTP ${res.status}`);
   641→    const buf = Buffer.from(await res.arrayBuffer());
   642→    fs.writeFileSync(zipPath, buf);
   643→    execPythonUnzip(zipPath, path.join(TMP_DIR, "xray"));

   645→    return true;
   646→  } catch (e) {
   647→    console.log(`xray download failed: ${e.message} — verify skipped`);
   648→    return false;
   649→  }
