   119→   119→function b64d(s) {
   120→   120→  try {
   121→   121→    let t = s.replace(/-/g, "+").replace(/_/g, "/").trim();
   122→   122→    while (t.length % 4) t += "=";
   123→   123→    const buf = Buffer.from(t, "base64");
   124→   124→    if (buf.length === 0) return "";
   125→   125→    const str = buf.toString("utf8");
   126→   126→    return /^[\x20-\x7e\s]/.test(str.slice(0, 8)) ? str : "";
   127→   127→  } catch {
   128→   128→    return "";
   129→   129→  }
   130→   130→}

   160→   160→function parseUserinfo(uri, proto) {
   161→   161→  const m = uri.match(
   162→   162→    new RegExp("^" + proto + "://[^@]+@([^:/#]+):(\\d+)(?:/[^?]*)?(?:\\?([^#]*))?(?:#(.*))?$")
   163→   163→  );
   164→   164→  if (!m) return null;
   165→   165→  const q = {};
   166→   166→  if (m[3])
   167→   167→    for (const kv of m[3].split("&")) {
   168→   168→      const i = kv.indexOf("=");
   169→   169→      if (i > 0) q[kv.slice(0, i)] = safeDecode(kv.slice(i + 1));
   170→   170→    }
   171→   171→  return {
   172→   172→    raw: uri,
   173→   173→    proto,
   174→   174→    host: m[1],
   175→   175→    port: Number(m[2]),
   176→   176→    name: safeDecode(m[4] || ""),
   177→   177→    net: q.type || q.network || "tcp",
   178→   178→    security: q.security || (q.tls === "1" ? "tls" : "none"),
   179→   179→  };
   180→   180→}

   190→   190→      name: safeDecode(m[4] || ""),
   191→   191→      net: "tcp",
   192→   192→      security: "none",
   193→   193→    };
   194→   194→  m = uri.match(/^ss:\/\/([^#]+?)(?:#(.*))?$/);

   201→   201→      proto: "ss",
   202→   202→      host: mm[1],
   203→   203→      port: Number(mm[2]),
   204→   204→      name: safeDecode(m[2] || ""),
   205→   205→      net: "tcp",
   206→   206→      security: "none",

   239→   239→function extractUris(text, uris) {
   240→   240→  for (const line of text.split(/\r?\n/)) {
   241→   241→    const t = line.trim();
   242→   242→    if (!t) continue;
   243→   243→    if (URI_RE.test(t)) {
   244→   244→      uris.add(t);
   245→   245→      continue;
   246→   246→    }
   247→   247→    const dec = b64d(t);
   248→   248→    if (dec && URI_RE.test(dec.trim())) uris.add(dec.trim());
   249→   249→  }

   285→   285→async function resolveHosts(cands) {
   286→   286→  const resolver = new Resolver();
   287→   287→  const deFirst = cands.filter((c) => remarkToCc(c.name));
   288→   288→  const others = cands.filter((c) => !remarkToCc(c.name));
   289→   289→  const hosts = [];
   290→   290→  const seen = new Set();
   291→   291→  for (const c of [...deFirst, ...others]) {
   292→   292→    const k = c.host.toLowerCase();
   293→   293→    if (seen.has(k)) continue;
   294→   294→    seen.add(k);
   295→   295→    hosts.push(c.host);
   296→   296→    if (hosts.length >= MAX_HOSTS_RESOLVE) break;
   297→   297→  }
   298→   298→  const ipToHosts = new Map();
   299→   299→  let idx = 0;
   300→   300→  const CONC = 60;
   301→   301→  const worker = async () => {
   302→   302→    while (idx < hosts.length) {
   303→   303→      const host = hosts[idx++];
   304→   304→      try {
   305→   305→        const addrs = await resolver.resolve4(host, { ttl: 300 });
   306→   306→        for (const a of addrs.slice(0, 3)) {
   307→   307→          if (!ipToHosts.has(a)) ipToHosts.set(a, []);
   308→   308→          ipToHosts.get(a).push(host.toLowerCase());
   309→   309→        }
   310→   310→      } catch {
   311→   311→        /* رزول نشد */
   312→   312→      }
   313→   313→    }
   314→   314→  };
   315→   315→  await Promise.all(Array.from({ length: CONC }, worker));
   316→   316→  return ipToHosts;
   317→   317→}

   319→   319→async function geoBatch(ipToHosts) {
   320→   320→  const ipList = [...ipToHosts.keys()].slice(0, MAX_GEO_IPS);
   321→   321→  const geo = new Map();

   324→   324→  const curlBatch = (body) => {

   326→   326→      const out = execFileSync(

   328→   328→        [
   329→   329→          "-sS", "--max-time", "20", "-X", "POST",

   333→   333→        ],
   334→   334→        { timeout: 25000, maxBuffer: 10 * 1024 * 1024 }
   335→   335→      );
   336→   336→      return JSON.parse(out.toString());
   337→   337→    } catch {
   338→   338→      return null;
   339→   339→    }
   340→   340→  };
   341→   341→  for (let i = 0; i < ipList.length && healthy; i += 100) {
   342→   342→    const chunk = ipList.slice(i, i + 100);
   343→   343→    let ok = false;
   344→   344→    for (let attempt = 0; attempt < 2 && !ok; attempt++) {
   345→   345→      const arr = curlBatch(JSON.stringify(chunk));
   346→   346→      if (Array.isArray(arr)) {
   347→   347→        for (const r of arr)
   348→   348→          if (r && r.query)
   349→   349→            geo.set(r.query, { cc: r.countryCode || "", org: r.org || r.isp || "" });
   350→   350→        ok = true;
   351→   351→      } else {
   352→   352→        await new Promise((r) => setTimeout(r, 5000));
   353→   353→      }
   354→   354→    }
   355→   355→    if (!ok) {
   356→   356→      if (geo.size === 0) healthy = false; // geo کلا در دسترس نیست
   357→   357→      break;
   358→   358→    }
   359→   359→    await new Promise((r) => setTimeout(r, 4300));
   360→   360→  }
   361→   361→  return { geo, healthy };
   362→   362→}

   365→   365→async function tcpAlive(cands) {
   366→   366→  const alive = [];
   367→   367→  let idx = 0;
   368→   368→  const CONC = 50;
   369→   369→  const tcpCheck = (host, port) =>
   370→   370→    new Promise((resolve) => {
   371→   371→      let settled = false;
   372→   372→      let s;
   373→   373→      const done = (ok) => {
   374→   374→        if (settled) return;
   375→   375→        settled = true;
   376→   376→        try {
   377→   377→          s.destroy();
   378→   378→        } catch {}
   379→   379→        resolve(ok);
   380→   380→      };
   381→   381→      try {
   382→   382→        s = net.connect({ host, port });
   383→   383→      } catch {
   384→   384→        return resolve(false);
   385→   385→      }
   386→   386→      s.setTimeout(3000, () => done(false));
   387→   387→      s.once("connect", () => done(true));
   388→   388→      s.once("error", () => done(false));
   389→   389→    });
   390→   390→  const worker = async () => {
   391→   391→    while (idx < cands.length) {
   392→   392→      const c = cands[idx++];
   393→   393→      if (await tcpCheck(c.host, c.port)) alive.push(c);
   394→   394→    }
   395→   395→  };
   396→   396→  await Promise.all(Array.from({ length: CONC }, worker));
   397→   397→  return alive;
   398→   398→}

   401→   401→function streamFromVmess(j, host) {
   402→   402→  const netMap = { raw: "tcp", splithttp: "xhttp", h2: "http" };
   403→   403→  const network = netMap[String(j.net || "tcp").toLowerCase()] || String(j.net || "tcp").toLowerCase();
   404→   404→  const s = { network, security: String(j.tls) === "tls" ? "tls" : "none" };
   405→   405→  if (s.security === "tls")
   406→   406→    s.tlsSettings = { serverName: String(j.sni || j.host || host), fingerprint: "chrome" };
   407→   407→  const p = String(j.path || "/");
   408→   408→  const h = String(j.host || "");
   409→   409→  if (network === "ws") s.wsSettings = { path: p, headers: h ? { Host: h } : undefined };
   410→   410→  else if (network === "grpc") s.grpcSettings = { serviceName: p.replace(/^\//, "") || "grpc" };
   411→   411→  else if (network === "httpupgrade") s.httpupgradeSettings = { path: p, host: h || host };
   412→   412→  else if (network === "xhttp") s.xhttpSettings = { path: p, host: h || host, mode: "auto" };
   413→   413→  else if (network === "http") s.httpSettings = { path: p, host: h ? [h] : undefined };
   414→   414→  else if (network === "tcp" && String(j.type) === "http")
   415→   415→    s.tcpSettings = {
   416→   416→      header: {
   417→   417→        type: "request",
   418→   418→        request: {
   419→   419→          version: "1.1",
   420→   420→          method: "GET",
   421→   421→          path: [p],
   422→   422→          headers: { Host: [h || host], "User-Agent": ["Mozilla/5.0"] },
   423→   423→        },
   424→   424→      },
   425→   425→    };
   426→   426→  return s;
   427→   427→}

   429→   429→function streamFromQuery(q, host, defaultTls) {
   430→   430→  const netMap = { raw: "tcp", splithttp: "xhttp" };
   431→   431→  const network = netMap[String(q.type || "tcp").toLowerCase()] || String(q.type || "tcp").toLowerCase();
   432→   432→  const sec = q.security || (defaultTls ? "tls" : "none");
   433→   433→  const s = { network, security: sec };
   434→   434→  if (sec === "tls")
   435→   435→    s.tlsSettings = { serverName: q.sni || q.host || q.peer || host, fingerprint: q.fp || "chrome" };
   436→   436→  else if (sec === "reality")
   437→   437→    s.realitySettings = {
   438→   438→      serverName: q.sni || host,
   439→   439→      publicKey: q.pbk || "",
   440→   440→      shortId: q.sid || "",
   441→   441→      fingerprint: q.fp || "chrome",
   442→   442→    };
   443→   443→  const p = safeDecode(q.path || "/");
   444→   444→  const h = q.host || "";
   445→   445→  if (network === "ws") s.wsSettings = { path: p, headers: h ? { Host: h } : undefined };
   446→   446→  else if (network === "grpc") s.grpcSettings = { serviceName: (q.serviceName || "").replace(/^\//, "") || "grpc" };
   447→   447→  else if (network === "httpupgrade") s.httpupgradeSettings = { path: p, host: h || host };
   448→   448→  else if (network === "xhttp") s.xhttpSettings = { path: p, host: h || host, mode: "auto" };
   449→   449→  else if (network === "http") s.httpSettings = { path: p, host: h ? [h] : undefined };
   450→   450→  else if (network === "tcp" && q.headerType === "http")
   451→   451→    s.tcpSettings = {
   452→   452→      header: { type: "request", request: { version: "1.1", method: "GET", path: [p], headers: { Host: [h || host] } } },
   453→   453→    };
   454→   454→  return s;
   455→   455→}

   457→   457→function buildOutbound(c) {
   458→   458→  const low = c.raw.slice(0, 8).toLowerCase();
   459→   459→  if (low.startsWith("vmess://")) {
   460→   460→    const j = JSON.parse(b64d(c.raw.slice(8)));
   461→   461→    return {
   462→   462→      protocol: "vmess",
   463→   463→      settings: {
   464→   464→        vnext: [{ address: c.host, port: c.port, users: [{ id: String(j.id), security: String(j.scy || "auto"), level: 0 }] }],
   465→   465→      },
   466→   466→      streamSettings: streamFromVmess(j, c.host),
   467→   467→    };
   468→   468→  }
   469→   469→  if (low.startsWith("vless://")) {
   470→   470→    const m = c.raw.match(/^vless:\/\/([^@]+)@([^:/#]+):(\d+)(?:\?([^#]*))?(?:#(.*))?$/);
   471→   471→    if (!m) return null;
   472→   472→    const q = {};
   473→   473→    if (m[4])
   474→   474→      for (const kv of m[4].split("&")) {
   475→   475→        const i = kv.indexOf("=");
   476→   476→        if (i > 0) q[kv.slice(0, i)] = safeDecode(kv.slice(i + 1));
   477→   477→      }
   478→   478→    return {
   479→   479→      protocol: "vless",
   480→   480→      settings: { vnext: [{ address: c.host, port: c.port, users: [{ id: m[1], encryption: "none", flow: q.flow || "", level: 0 }] }] },
   481→   481→      streamSettings: streamFromQuery(q, c.host, false),
   482→   482→    };
   483→   483→  }
   484→   484→  if (low.startsWith("trojan://")) {
   485→   485→    const m = c.raw.match(/^trojan:\/\/([^@]+)@([^:/#]+):(\d+)(?:\?([^#]*))?(?:#(.*))?$/);
   486→   486→    if (!m) return null;
   487→   487→    const q = {};
   488→   488→    if (m[4])
   489→   489→      for (const kv of m[4].split("&")) {
   490→   490→        const i = kv.indexOf("=");
   491→   491→        if (i > 0) q[kv.slice(0, i)] = safeDecode(kv.slice(i + 1));
   492→   492→      }
   493→   493→    return {
   494→   494→      protocol: "trojan",
   495→   495→      settings: { servers: [{ address: c.host, port: c.port, password: safeDecode(m[1]) }] },
   496→   496→      streamSettings: streamFromQuery(q, c.host, q.security !== "none"),
   497→   497→    };
   498→   498→  }
   499→   499→  if (low.startsWith("ss://")) {
   500→   500→    let userinfo = "";
   501→   501→    const m1 = c.raw.match(/^ss:\/\/([^@#]+)@/);
   502→   502→    if (m1) userinfo = b64d(m1[1]);
   503→   503→    else {
   504→   504→      const m2 = c.raw.match(/^ss:\/\/([^#]+)/);
   505→   505→      const dec = b64d(m2 ? m2[1] : "");
   506→   506→      const mi = dec.indexOf("@");
   507→   507→      userinfo = mi > 0 ? dec.slice(0, mi) : "";
   508→   508→    }
   509→   509→    const ci = userinfo.indexOf(":");
   510→   510→    if (ci < 0) return null;
   511→   511→    return {
   512→   512→      protocol: "shadowsocks",
   513→   513→      settings: {
   514→   514→        servers: [{ address: c.host, port: c.port, method: userinfo.slice(0, ci), password: userinfo.slice(ci + 1) }],
   515→   515→      },
   516→   516→    };
   517→   517→  }
   518→   518→  return null;
   519→   519→}

   576→   576→async function verifyBatch(list, expectedCc) {
   577→   577→  const verified = new Set();
   578→   578→  if (list.length === 0) return verified;
   579→   579→  let idx = 0;
   580→   580→  const N = Math.min(PARALLEL_XRAY, list.length);
   581→   581→  const freeSlots = [...Array(N).keys()];
   582→   582→  const takeSlot = () =>
   583→   583→    new Promise((res) => {
   584→   584→      const t = setInterval(() => {
   585→   585→        if (freeSlots.length > 0) {
   586→   586→          clearInterval(t);
   587→   587→          res(freeSlots.shift());
   588→   588→        }
   589→   589→      }, 30);
   590→   590→    });
   591→   591→  const worker = async () => {
   592→   592→    while (idx < list.length) {
   593→   593→      const c = list[idx++];
   594→   594→      const slot = await takeSlot();
   595→   595→      try {
   596→   596→        const ok = await verifyOne(c, slot, expectedCc);
   597→   597→        if (ok) verified.add(c.raw);
   598→   598→      } catch {
   599→   599→        /* نادیده */
   600→   600→      }
   601→   601→      freeSlots.push(slot);
   602→   602→    }
   603→   603→  };
   604→   604→  await Promise.all(Array.from({ length: N }, worker));
   605→   605→  return verified;
   606→   606→}

   609→   609→function score(c) {
   610→   610→  let s = 0;
   611→   611→  if (PREF_PORTS.has(c.port)) s += 4;
   612→   612→  if (c.security === "reality") s += 3;

   614→   614→  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(c.host)) s += 1; // IP مستقیم

   617→   617→function pickForVerify(pool, n) {
   618→   618→  const byProto = new Map();
   619→   619→  const chosen = [];
   620→   620→  const sorted = [...pool].sort((a, b) => score(b) - score(a));
   621→   621→  for (const c of sorted) {
   622→   622→    const k = `${c.proto}/${c.security}/${c.net}`;
   623→   623→    const cnt = byProto.get(k) || 0;
   624→   624→    if (cnt >= Math.ceil(n / 4) && chosen.length > 0) continue;
   625→   625→    byProto.set(k, cnt + 1);
   626→   626→    chosen.push(c);
   627→   627→    if (chosen.length >= n) break;
   628→   628→  }
   629→   629→  return chosen;
   630→   630→}
